# SonRac (Sonic & All-Stars Racing Transformed) - Android native build.
#
# Builds libmain.so (recompiled game + app layer) against the Android-capable
# ReXGlue SDK from LibertyRecompAndroid, which also builds librexruntime.so,
# librexgpu-xenos.so and libSDL3.so. SDL's SDLActivity loads libmain.so and
# calls SDL_main. Mirrors gta4-recomp's Android target, without the GTA IV
# specific renderer: Sonic uses the generic Xenos GPU plugin.
#
# Configure (Git Bash):
#   cmake -S android_native -B out/android-arm64 -G Ninja \
#     -DCMAKE_TOOLCHAIN_FILE=$NDK/build/cmake/android.toolchain.cmake \
#     -DANDROID_ABI=arm64-v8a -DANDROID_PLATFORM=android-28 -DANDROID_STL=c++_shared \
#     -DCMAKE_C_FLAGS="-march=armv8.2-a -fno-emulated-tls" \
#     -DCMAKE_CXX_FLAGS="-march=armv8.2-a -fno-emulated-tls" \
#     -DCMAKE_BUILD_TYPE=RelWithDebInfo -DREXGLUE_ENABLE_TRACY=OFF

cmake_minimum_required(VERSION 3.25)
project(sonrac LANGUAGES C CXX)

if(NOT ANDROID)
    message(FATAL_ERROR "android_native is only for Android builds (use the top-level CMakeLists.txt for PC)")
endif()

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(SONRAC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(REXSDK_DIR "C:/Users/EduPedu/Desktop/LibertyRecompAndroid/glue/rexglue-sdk-main"
    CACHE PATH "Android-capable rexglue-sdk source tree")

set(REXGLUE_BUILD_GTA4_RECOMP OFF CACHE BOOL "" FORCE)
set(REXGLUE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(REXGLUE_BUILD_GTA4_NATIVE OFF CACHE BOOL "" FORCE)
set(REXGLUE_OUTPUT_DIR "${CMAKE_BINARY_DIR}/rex-out" CACHE PATH "" FORCE)
add_subdirectory("${REXSDK_DIR}" rexglue-sdk)

# Generated sources list (paths relative to the project root)
set(_sonrac_saved_dir "${CMAKE_CURRENT_SOURCE_DIR}")
include("${SONRAC_ROOT}/generated/sources.cmake")
set(SONRAC_GENERATED_SOURCES "")
foreach(src IN LISTS GENERATED_SOURCES)
    if(IS_ABSOLUTE "${src}")
        list(APPEND SONRAC_GENERATED_SOURCES "${src}")
    else()
        list(APPEND SONRAC_GENERATED_SOURCES "${SONRAC_ROOT}/${src}")
    endif()
endforeach()

add_library(SonRac SHARED
    ${SONRAC_ROOT}/src/main.cpp
    ${SONRAC_ROOT}/src/android_virtual_pad.cpp
    ${SONRAC_ROOT}/src/android_power.cpp
    ${SONRAC_ROOT}/src/android_d3d_waits.cpp
    ${SONRAC_ROOT}/src/native_video_convert.cpp
    ${SONRAC_ROOT}/src/cri_frame_pools.cpp
    ${SONRAC_ROOT}/src/frame_pacing_probe.cpp
    ${SONRAC_ROOT}/src/cri_wakeup_probe.cpp
    ${SONRAC_ROOT}/src/shader_dump.cpp
    ${SONRAC_GENERATED_SOURCES}
)

# El renderizador nativo se reconstruira desde cero para SonRac (sin herencia de otros juegos).

# Phase A of the native renderer: count the calls into the Direct3D zone
# (src/android_d3d_trace.cpp, generated by tools/gen_d3d_trace.py).
option(SONRAC_D3D_TRACE "Build the Direct3D call tracer" OFF)
if(SONRAC_D3D_TRACE)
    target_sources(SonRac PRIVATE ${SONRAC_ROOT}/src/android_d3d_trace.cpp)
endif()

target_include_directories(SonRac PRIVATE
    ${SONRAC_ROOT}
    ${SONRAC_ROOT}/src
    ${SONRAC_ROOT}/generated
    ${REXSDK_DIR}/thirdparty/imgui
    ${REXSDK_DIR}/src/ui/shaders
    ${REXSDK_DIR}/thirdparty/xxHash
    ${REXSDK_DIR}/thirdparty/renderdoc
)

target_link_libraries(SonRac PRIVATE
    rex::runtime
    SDL3::SDL3
    z
    android
    log
)

rexglue_configure_target(SonRac GPU_PLUGINS xenos)

set_target_properties(SonRac PROPERTIES OUTPUT_NAME "main")
# SDL resolves SDL_main with dlsym after dlopen; Android 15+ devices need
# 16 KB-aligned segments.
target_link_options(SonRac PRIVATE
    "-Wl,-z,max-page-size=16384"
    "-Wl,--export-dynamic-symbol=SDL_main"
    "-Wl,--export-dynamic-symbol=JNI_OnLoad"
    "-Wl,--export-dynamic-symbol=Java_com_sonrac_game_*"
    "-Wl,-Bsymbolic-functions"
)
# The recompiled game at -O3 with ThinLTO, as LibertyRecompAndroid builds
# libmain: calls between the ~100k small recompiled functions can then be
# inlined across translation units at link time.
target_compile_options(SonRac PRIVATE -O3 -Wno-everything)
option(SONRAC_ANDROID_THINLTO "ThinLTO for libmain on Android" ON)
if(SONRAC_ANDROID_THINLTO)
    target_compile_options(SonRac PRIVATE -flto=thin)
    target_link_options(SonRac PRIVATE -flto=thin "-Wl,--thinlto-jobs=8"
        "-Wl,--thinlto-cache-dir=${CMAKE_BINARY_DIR}/thinlto-cache")
endif()
